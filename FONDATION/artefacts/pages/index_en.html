Reproduce the calculation

# from the repository root
py -3 FONDATION/fondation_base.py                 # 18 equations, 18 checks
py -3 FONDATION/generer_fondation.py              # rebuilds pages, figure, JSON, CSV, Markdown
py -3 -m unittest FONDATION.tests.test_fondation  # 45 tests
py -3 FONDATION/OUTILS/audit_piliers.py FONDATION/fondation_base.py --rigide
# off Windows: replace py -3 with python3
